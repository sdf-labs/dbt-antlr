   fn handleAcceptPositionForKeyword<'input, 'arena, Input:CharStream<'input>, TF: TokenFactory<'input, 'arena> + 'arena>(this: &mut BaseLexerType<'input, 'arena, Input, TF>, keyword: &str) {

   fn isIdentifierChar(c: char) -> bool{
